    // Extract YouTube Video ID
    function getYouTubeEmbedUrl(url) {
      if (!url) return "";
      const regExp = /^.*(youtu.be\/|v\/|u\/\w\/|embed\/|watch\?v=|\&v=)([^#\&\?]*).*/;
      const match = url.match(regExp);
      if (match && match[2].length === 11) {
        return `https://www.youtube.com/embed/${match[2]}?autoplay=1&rel=0`;
      }
      return url;
    }

    // Render Main Content
    function renderContent() {
      const container = document.getElementById("mainContent");
      const query = document.getElementById("searchInput").value.trim().toLowerCase();

      // If searching or specific category selected -> Grid View
      if (query || selectedCategory !== "All") {
        const filtered = appData.movies.filter(m => {
          const matchesCat = selectedCategory === "All" || m.category === selectedCategory;
          const matchesQuery = !query || m.title.toLowerCase().includes(query) || m.category.toLowerCase().includes(query);
          return matchesCat && matchesQuery;
        });

        container.innerHTML = `
          <div>
            <div class="flex items-center justify-between mb-4">
              <h2 class="text-xl font-bold flex items-center gap-2">
                <span class="w-1.5 h-6 brand-bg rounded-full inline-block"></span>
                ${query ? `Search Results for "${query}"` : selectedCategory} (${filtered.length})
              </h2>
              <button onclick="resetFilters()" class="text-xs brand-text hover:underline cursor-pointer">Show All</button>
            </div>
            ${filtered.length === 0
              ? `<div class="text-center py-16 text-gray-500">Koi movie ya series nahi mili. Customize button se nayi add karein!</div>`
              : `<div class="flex flex-wrap gap-4 sm:gap-5">${filtered.map(createMovieCard).join("")}</div>`
            }
          </div>
        `;
        return;
      }

      // Default Netflix-style Rows by Category
      let html = "";
      appData.categories.forEach(cat => {
        const catMovies = appData.movies.filter(m => m.category === cat);
        if (catMovies.length > 0) {
          html += `
            <div class="space-y-3">
              <div class="flex items-center justify-between">
                <h2 class="text-lg sm:text-xl font-bold flex items-center gap-2">
                  <span class="w-1.5 h-5 brand-bg rounded-full inline-block"></span>
                  ${cat}
                </h2>
                <button onclick="selectCategory('${cat}')" class="text-xs text-gray-400 hover:text-white cursor-pointer">See All &rarr;</button>
              </div>
              <div class="flex gap-4 overflow-x-auto no-scrollbar pb-2">
                ${catMovies.map(createMovieCard).join("")}
              </div>
            </div>
          `;
        }
      });

      container.innerHTML = html || `<div class="text-center py-16 text-gray-500">Abhi koi content nahi hai. Top-right "Customize" button se add karein.</div>`;
    }

    // Save / Update Movie
    function saveMovie(e) {
      e.preventDefault();
      const editId = document.getElementById("editMovieId").value;
      const movieObj = {
        id: editId || "m" + Date.now(),
        title: document.getElementById("mTitle").value.trim(),
        category: document.getElementById("mCategory").value,
        type: document.getElementById("mType").value,
        poster: document.getElementById("mPoster").value.trim(),
        backdrop: document.getElementById("mBackdrop").value.trim(),
        youtube: document.getElementById("mYoutube").value.trim(),
        year: document.getElementById("mYear").value.trim() || "2026",
        rating: document.getElementById("mRating").value.trim() || "8.0",
        watchLink: document.getElementById("mWatchLink").value.trim(),
        desc: document.getElementById("mDesc").value.trim()
      };

      if (editId) {
        const idx = appData.movies.findIndex(m => m.id === editId);
        if (idx !== -1) appData.movies[idx] = movieObj;
      } else {
        appData.movies.unshift(movieObj);
      }

      saveDataToStorage();
      resetMovieForm();
      renderAdminMovies();
      populateAdminDropdowns();
      alert(editId ? "Updated successfully!" : "Added to Flimy World!");
    }

      document.getElementById("mTitle").value = m.title;
      document.getElementById("mCategory").value = m.category;

    // Backup & Restore
    function exportData() {
      const blob = new Blob([JSON.stringify(appData, null, 2)], { type: "application/json" });
      const url = URL.createObjectURL(blob);
      const a = document.createElement("a");
      a.href = url;
      a.download = "flimy-world-backup.json";
      a.click();
    }

    function importData(e) {
      const file = e.target.files[0];
      if (!file) return;
      const reader = new FileReader();
      reader.onload = function(event) {
        try {
          const parsed = JSON.parse(event.target.result);
          if (parsed.movies && parsed.categories) {
            appData = parsed;
            saveDataToStorage();
            openAdminModal();
            alert("Backup restored successfully!");
          } else {
            alert("Invalid JSON file.");
          }
        } catch (err) {
          alert("File read error.");
        }
      };
      reader.readAsText(file);
    }
